        package com.ravi.consumer;
        import java.util.function.Consumer;

        record Employee(int id, String name, double salary) { }

        public class ConsumerDemo1 {
            public static void main(String[] args) {
                Consumer<Integer> c1 = num -> IO.println("Integer type is :"+num);
                c1.accept(12);

                Consumer<Employee> c2 = emp -> {
                    IO.println("Employee Id is :"+emp.id());
                    IO.println("Employee Name is :"+emp.name());
                    IO.println("Employee Salary is :"+emp.salary());
                };
                c2.accept(new Employee(111, "Scott", 800));
            }
        }
